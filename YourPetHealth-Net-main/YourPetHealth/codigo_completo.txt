using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;
using Oracle.EntityFrameworkCore.Infrastructure;
using OpenTelemetry;
using OpenTelemetry.Metrics;
using OpenTelemetry.Resources;
using OpenTelemetry.Trace;
using Serilog;
using YourPetHealth.Data;
using YourPetHealth.Diagnostics;
using YourPetHealth.HealthChecks;
using YourPetHealth.Middleware;
using YourPetHealth.Models;
using YourPetHealth.Services; // <-- Adicionado para reconhecer o MongoDbService

Log.Logger = new LoggerConfiguration()
    .WriteTo.Console()
    .CreateBootstrapLogger();

try
{
    var builder = WebApplication.CreateBuilder(args);

    builder.Host.UseSerilog((context, services, configuration) => configuration
        .ReadFrom.Configuration(context.Configuration)
        .ReadFrom.Services(services)
        .Enrich.FromLogContext()
        .Enrich.WithMachineName()
        .Enrich.WithThreadId());

    builder.Services.AddControllersWithViews();

    // Registo do Swagger / OpenAPI
    builder.Services.AddEndpointsApiExplorer();
    builder.Services.AddSwaggerGen();

    // Registo do ServiÃ§o do MongoDB para a Telemetria
    builder.Services.AddSingleton<MongoDbService>();

    builder.Services.AddDbContext<AppDbContext>(options =>
        options.UseOracle(
            builder.Configuration.GetConnectionString("OracleConnection"),
            oracleOptions => oracleOptions.UseOracleSQLCompatibility(OracleSQLCompatibility.DatabaseVersion19)
        )
    );

    builder.Services.AddScoped<IPasswordHasher<Usuario>, PasswordHasher<Usuario>>();

    builder.Services
        .AddAuthentication(CookieAuthenticationDefaults.AuthenticationScheme)
        .AddCookie(options =>
        {
            options.LoginPath = "/Auth/Login";
            options.AccessDeniedPath = "/Auth/Login";
            options.ExpireTimeSpan = TimeSpan.FromHours(8);
            options.SlidingExpiration = true;
        });

    builder.Services
        .AddHealthChecks()
        .AddCheck<OracleDatabaseHealthCheck>(
            "oracle_database",
            tags: new[] { "db", "oracle", "ready" })
        .AddCheck<ExternalServiceHealthCheck>(
            "external_service",
            tags: new[] { "external", "ready" });

    builder.Services.AddHttpClient<ExternalServiceHealthCheck>();

    builder.Services.AddOpenTelemetry()
        .ConfigureResource(resource => resource.AddService(
            serviceName: AppDiagnostics.ServiceName,
            serviceVersion: "1.0.0"))
        .WithTracing(tracing => tracing
            .AddSource(AppDiagnostics.ServiceName)
            .AddAspNetCoreInstrumentation()
            .AddHttpClientInstrumentation()
            .AddConsoleExporter())
        .WithMetrics(metrics => metrics
            .AddMeter(AppDiagnostics.ServiceName)
            .AddAspNetCoreInstrumentation()
            .AddHttpClientInstrumentation()
            .AddConsoleExporter());

    var app = builder.Build();

    app.UseSerilogRequestLogging();

    app.UseMiddleware<CorrelationIdMiddleware>();

    if (!app.Environment.IsDevelopment())
    {
        app.UseExceptionHandler("/Home/Error");
    }
    else
    {
        // AtivaÃ§Ã£o do Swagger no ambiente de desenvolvimento
        app.UseSwagger();
        app.UseSwaggerUI();
    }

    app.UseStaticFiles();

    app.UseRouting();

    app.UseAuthentication();
    app.UseAuthorization();

    app.MapHealthChecks("/health", new Microsoft.AspNetCore.Diagnostics.HealthChecks.HealthCheckOptions
    {
        ResponseWriter = HealthCheckResponseWriter.WriteResponse
    });

    app.MapHealthChecks("/health/ready", new Microsoft.AspNetCore.Diagnostics.HealthChecks.HealthCheckOptions
    {
        Predicate = check => check.Tags.Contains("ready"),
        ResponseWriter = HealthCheckResponseWriter.WriteResponse
    });

    app.MapControllerRoute(
        name: "default",
        pattern: "{controller=Home}/{action=Index}/{id?}");

    app.Run();
}
catch (Exception ex) when (ex.GetType().Name != "StopTheHostException")
{
    Log.Fatal(ex, "A aplicaÃ§Ã£o encerrou de forma inesperada durante a inicializaÃ§Ã£o");
}
finally
{
    Log.CloseAndFlush();
}

        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Login(string email, string senha, string? returnUrl = null)
        {
            using var activity = AppDiagnostics.ActivitySource.StartActivity("Auth.Login");
            activity?.SetTag("auth.email", email);

            ViewData["ReturnUrl"] = returnUrl;

            if (string.IsNullOrWhiteSpace(email) || string.IsNullOrWhiteSpace(senha))
            {
                ModelState.AddModelError(string.Empty, "Informe email e senha.");
                return View();
            }

            var usuario = await _context.Usuarios
                .Include(u => u.Responsavel)
                .Include(u => u.Veterinario)
                .FirstOrDefaultAsync(u => u.Email == email && u.Ativo);

            if (usuario == null)
            {
                AppDiagnostics.LoginsFalhos.Add(1);
                _logger.LogWarning("Tentativa de login falhou: email {Email} nÃ£o encontrado ou inativo", email);
                ModelState.AddModelError(string.Empty, "Email ou senha invÃ¡lidos.");
                return View();
            }

            var resultado = _passwordHasher.VerifyHashedPassword(usuario, usuario.SenhaHash, senha);
            if (resultado == PasswordVerificationResult.Failed)
            {
                AppDiagnostics.LoginsFalhos.Add(1);
                _logger.LogWarning("Tentativa de login falhou: senha incorreta para {Email}", email);
                ModelState.AddModelError(string.Empty, "Email ou senha invÃ¡lidos.");
                return View();
            }

            var nomeExibicao =
                usuario.Responsavel?.Nome
                ?? usuario.Veterinario?.Nome
                ?? usuario.Email;

            var claims = new List<Claim>
            {
                new(ClaimTypes.NameIdentifier, usuario.Id.ToString()),
                new(ClaimTypes.Name, nomeExibicao),
                new(ClaimTypes.Email, usuario.Email),
                new(ClaimTypes.Role, usuario.Perfil)
            };

            if (usuario.ResponsavelId.HasValue)
                claims.Add(new Claim("EntidadeId", usuario.ResponsavelId.Value.ToString()));
            if (usuario.VeterinarioId.HasValue)
                claims.Add(new Claim("EntidadeId", usuario.VeterinarioId.Value.ToString()));

            var identity = new ClaimsIdentity(claims, CookieAuthenticationDefaults.AuthenticationScheme);
            await HttpContext.SignInAsync(CookieAuthenticationDefaults.AuthenticationScheme,
                new ClaimsPrincipal(identity));

            AppDiagnostics.LoginsRealizados.Add(1);
            activity?.SetTag("auth.perfil", usuario.Perfil);
            _logger.LogInformation("Login bem-sucedido para {Email} (perfil {Perfil})", email, usuario.Perfil);

            if (!string.IsNullOrEmpty(returnUrl) && Url.IsLocalUrl(returnUrl))
                return Redirect(returnUrl);

            return usuario.Perfil switch
            {
                PerfilUsuario.Veterinario => RedirectToAction("Index", "Responsaveis"),
                PerfilUsuario.Responsavel => RedirectToAction("Index", "Pets"),
                _ => RedirectToAction("Index", "Home")
            };
        }

        [AllowAnonymous]
        public IActionResult Register() => View();

        [AllowAnonymous]
        [HttpPost]
        public async Task<IActionResult> Register(RegisterViewModel model)
        {
            if (model.TipoConta != PerfilUsuario.Responsavel && model.TipoConta != PerfilUsuario.Veterinario)
                ModelState.AddModelError(nameof(model.TipoConta), "Selecione o tipo de conta.");

            if (model.TipoConta == PerfilUsuario.Veterinario)
            {
                ModelState.Remove(nameof(model.Crmv));
                if (string.IsNullOrWhiteSpace(model.Crmv))
                    ModelState.AddModelError(nameof(model.Crmv), "CRMV Ã© obrigatÃ³rio para veterinÃ¡rios.");
            }
            else
            {
                ModelState.Remove(nameof(model.Crmv));
                ModelState.Remove(nameof(model.Especialidade));
            }

            if (!ModelState.IsValid) return View(model);

            using var transaction = await _context.Database.BeginTransactionAsync();
            int entidadeId;

            if (model.TipoConta == PerfilUsuario.Veterinario)
            {
                if (await _context.Veterinarios.AnyAsync(v => v.Email == model.Email))
                {
                    ModelState.AddModelError(nameof(model.Email), "Email jÃ¡ cadastrado.");
                    return View(model);
                }
                if (await _context.Veterinarios.AnyAsync(v => v.Crmv == model.Crmv))
                {
                    ModelState.AddModelError(nameof(model.Crmv), "CRMV jÃ¡ cadastrado.");
                    return View(model);
                }

                var veterinario = new Veterinario
                {
                    Nome = model.Nome,
                    Email = model.Email,
                    Telefone = model.Telefone,
                    Crmv = model.Crmv!,
                    Especialidade = model.Especialidade
                };
                _context.Veterinarios.Add(veterinario);
                await _context.SaveChangesAsync();
                entidadeId = veterinario.Id;
            }
            else
            {
                if (await _context.Responsaveis.AnyAsync(r => r.Email == model.Email))
                {
                    ModelState.AddModelError(nameof(model.Email), "Email jÃ¡ cadastrado.");
                    return View(model);
                }

                var responsavel = new Responsavel
                {
                    Nome = model.Nome,
                    Email = model.Email,
                    Telefone = model.Telefone
                };
                _context.Responsaveis.Add(responsavel);
                await _context.SaveChangesAsync();
                entidadeId = responsavel.Id;

                AppDiagnostics.CadastrosResponsavel.Add(1);
            }

            var usuario = new Usuario
            {
                Email = model.Email,
                Perfil = model.TipoConta,
                Ativo = true,
                CriadoEm = DateTime.UtcNow,
                ResponsavelId = model.TipoConta == PerfilUsuario.Responsavel ? entidadeId : null,
                VeterinarioId = model.TipoConta == PerfilUsuario.Veterinario ? entidadeId : null
            };
            usuario.SenhaHash = _passwordHasher.HashPassword(usuario, model.Senha);

            _context.Usuarios.Add(usuario);
            await _context.SaveChangesAsync();
            await transaction.CommitAsync();

            _logger.LogInformation("Conta criada para {Email} (perfil {Perfil})", model.Email, model.TipoConta);
            TempData["Sucesso"] = "Conta criada com sucesso! JÃ¡ pode entrar.";
            return RedirectToAction(nameof(Login));
        }

    [Authorize]
    public class ConsultasController : Controller
    {
        private readonly AppDbContext _context;
        public ConsultasController(AppDbContext context) => _context = context;

        public async Task<IActionResult> Index()
        {
            var query = _context.Consultas.Include(c => c.Pet).ThenInclude(p => p.Responsavel).Include(c => c.Veterinario).AsQueryable();

            var meuId = User.GetEntidadeId();
            if (User.IsInRole(PerfilUsuario.Responsavel))
                query = query.Where(c => c.Pet.ResponsavelId == meuId);
            else if (User.IsInRole(PerfilUsuario.Veterinario))
                query = query.Where(c => c.VeterinarioId == meuId);

            return View(await query.OrderByDescending(c => c.Data).ToListAsync());
        }

        [Authorize(Roles = "RESPONSAVEL")]
        public async Task<IActionResult> Create()
        {
            var meuId = User.GetEntidadeId();
            ViewBag.PetId = new SelectList(
                await _context.Pets.Where(p => p.ResponsavelId == meuId).ToListAsync(), "Id", "Nome");
            ViewBag.VeterinarioId = new SelectList(await _context.Veterinarios.ToListAsync(), "Id", "Nome");
            return View();
        }

        [Authorize(Roles = "RESPONSAVEL")]
        [HttpPost]
        public async Task<IActionResult> Create(Consulta c)
        {
            var meuId = User.GetEntidadeId();
            var meusPets = await _context.Pets.Where(p => p.ResponsavelId == meuId).ToListAsync();

            ViewBag.PetId = new SelectList(meusPets, "Id", "Nome", c.PetId);
            ViewBag.VeterinarioId = new SelectList(await _context.Veterinarios.ToListAsync(), "Id", "Nome", c.VeterinarioId);

            if (!meusPets.Any(p => p.Id == c.PetId))
            {
                ModelState.AddModelError("PetId", "Selecione um dos seus pets.");
                return View(c);
            }

            c.Status = "AGENDADA";
            ModelState.Remove(nameof(Consulta.Status));

            if (!ModelState.IsValid) return View(c);

            _context.Consultas.Add(c);
            await _context.SaveChangesAsync();

            TempData["Sucesso"] = "Consulta agendada com sucesso!";
            return RedirectToAction(nameof(Index));
        }

        [Authorize(Roles = "VETERINARIO,ADMIN")]
        public async Task<IActionResult> Edit(int id)
        {
            var c = await _context.Consultas.FindAsync(id);
            if (c == null) return NotFound();
            if (!ResponsavelPelaConsulta(c)) return Forbid();

            ViewBag.Petid = new SelectList(await _context.Pets.ToListAsync(), "Id", "Nome", c.PetId);
            ViewBag.VeterinarioId = new SelectList(await _context.Veterinarios.ToListAsync(), "Id", "Nome", c.VeterinarioId);
            return View(c);
        }

        [Authorize(Roles = "VETERINARIO,ADMIN")]
        [HttpPost]
        public async Task<IActionResult> Edit(Consulta atualizado)
        {
            var consulta = await _context.Consultas.FindAsync(atualizado.Id);
            if (consulta == null) return NotFound();
            if (!ResponsavelPelaConsulta(consulta)) return Forbid();

            ViewBag.PetId = new SelectList(await _context.Pets.ToListAsync(), "Id", "Nome", atualizado.PetId);
            ViewBag.VeterinarioId = new SelectList(await _context.Veterinarios.ToListAsync(), "Id", "Nome", atualizado.VeterinarioId);

            var status = atualizado.Status?.ToUpper() ?? "";
            if (status != "REALIZADA" && status != "CANCELADA")
            {
                ModelState.AddModelError("Status", "Status deve ser atualizado para Realizada ou Cancelada!");
                return View(atualizado);
            }

            consulta.Status = status;
            consulta.Observacoes = atualizado.Observacoes;

            await _context.SaveChangesAsync();

            TempData["Sucesso"] = "Consulta atualizada com sucesso!";
            return RedirectToAction(nameof(Index));
        }

        public async Task<IActionResult> Delete(int id)
        {
            var c = await _context.Consultas.Include(x => x.Pet).ThenInclude(p => p.Responsavel).Include(x => x.Veterinario).FirstOrDefaultAsync(x => x.Id == id);
            if (c == null) return NotFound();
            if (!PodeVerConsulta(c)) return Forbid();
            return View(c);
        }

        [HttpPost, ActionName("Delete")]
        public async Task<IActionResult> DeleteConfirmed(int id)
        {
            var c = await _context.Consultas.Include(x => x.Pet).FirstOrDefaultAsync(x => x.Id == id);
            if (c == null) return NotFound();
            if (!PodeVerConsulta(c)) return Forbid();

            _context.Consultas.Remove(c);
            await _context.SaveChangesAsync();

            TempData["Sucesso"] = "Consulta removida com sucesso!";
            return RedirectToAction(nameof(Index));
        }

        private bool ResponsavelPelaConsulta(Consulta c) =>
            User.IsInRole(PerfilUsuario.Admin) || c.VeterinarioId == User.GetEntidadeId();

        private bool PodeVerConsulta(Consulta c)
        {
            if (User.IsInRole(PerfilUsuario.Admin)) return true;
            if (User.IsInRole(PerfilUsuario.Veterinario)) return c.VeterinarioId == User.GetEntidadeId();
            if (User.IsInRole(PerfilUsuario.Responsavel)) return c.Pet.ResponsavelId == User.GetEntidadeId();
            return false;
        }
    }

        public async Task<IActionResult> Index()
        {
            if (User.IsInRole(PerfilUsuario.Responsavel))
            {
                var meuId = User.GetEntidadeId();
                var agora = DateTime.Now;

                var meusPets = await _context.Pets
                    .Where(p => p.ResponsavelId == meuId)
                    .ToListAsync();

                var proximasConsultas = await _context.Consultas
                    .Include(c => c.Pet)
                    .Include(c => c.Veterinario)
                    .Where(c => c.Pet.ResponsavelId == meuId && c.Status == "AGENDADA" && c.Data >= agora)
                    .OrderBy(c => c.Data)
                    .Take(5)
                    .ToListAsync();

                ViewBag.MeusPets = meusPets;
                ViewBag.ProximasConsultas = proximasConsultas;
                ViewBag.ConsultasProximas48h = proximasConsultas.Count(c => c.Data <= agora.AddHours(48));

                return View("Tutor");
            }

            ViewBag.TotalResponsaveis = await _context.Responsaveis.CountAsync();
            ViewBag.TotalVeterinarios = await _context.Veterinarios.CountAsync();
            ViewBag.TotalPets = await _context.Pets.CountAsync();
            ViewBag.TotalConsultas = await _context.Consultas.CountAsync();
            ViewBag.ConsultasAgendadas = await _context.Consultas.CountAsync(c => c.Status == "AGENDADA");

            return View();
        }

        public IActionResult Error()
        {

        {
            base.OnModelCreating(modelBuilder);

            modelBuilder.Entity<Responsavel>()
                .Property(r => r.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<Veterinario>()
                .Property(v => v.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<Pet>()
                .Property(p => p.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<Consulta>()
                .Property(c => c.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<HistoricoClinico>()
                .Property(h => h.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<Usuario>()
                .Property(u => u.Id)
                .ValueGeneratedOnAdd();

            modelBuilder.Entity<Responsavel>()
                .HasIndex(r => r.Email)
                .IsUnique()
                .HasDatabaseName("UQ_RESPONSAVEL_EMAIL");

            modelBuilder.Entity<Veterinario>()
                .HasIndex(v => v.Email)
                .IsUnique()
                .HasDatabaseName("UQ_VETERINARIO_EMAIL");

            modelBuilder.Entity<Veterinario>()
                .HasIndex(v => v.Crmv)
                .IsUnique()
                .HasDatabaseName("UQ_VETERINARIO_CRMV");

            if (Database.IsRelational())
            {
                modelBuilder.Entity<Pet>().ToTable(tb =>
                    tb.HasCheckConstraint("CK_PET_SEXO", "SEXO_PET IN ('MACHO','FEMEA')"));

                modelBuilder.Entity<Consulta>().ToTable(tb =>
                {
                    tb.HasCheckConstraint("CK_CONSULTA_STATUS", "STATUS IN ('AGENDADA','REALIZADA','CANCELADA')");
                    tb.HasCheckConstraint("CK_CONSULTA_TIPO", "TIPO IN ('ROTINA','VACINACAO','EMERGENCIA','RETORNO','EXAME')");
                });

                modelBuilder.Entity<Usuario>().ToTable(tb =>
                {
                    tb.HasCheckConstraint("CK_USUARIO_PERFIL",
                        "PERFIL IN ('RESPONSAVEL','VETERINARIO','ADMIN')");

                    tb.HasCheckConstraint("CK_USUARIO_VINCULO",
                        "(PERFIL = 'RESPONSAVEL' AND RESPONSAVEL_ID IS NOT NULL AND VETERINARIO_ID IS NULL) " +
                        "OR (PERFIL = 'VETERINARIO' AND VETERINARIO_ID IS NOT NULL AND RESPONSAVEL_ID IS NULL) " +
                        "OR (PERFIL = 'ADMIN' AND RESPONSAVEL_ID IS NULL AND VETERINARIO_ID IS NULL)");
                });
            }

            modelBuilder.Entity<Pet>()
                .HasOne(p => p.Responsavel)
                .WithMany(r => r.Pets)
                .HasForeignKey(p => p.ResponsavelId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Consulta>()
                .HasOne(c => c.Pet)
                .WithMany(p => p.Consultas)
                .HasForeignKey(c => c.PetId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Consulta>()
                .HasOne(c => c.Veterinario)
                .WithMany(v => v.Consultas)
                .HasForeignKey(c => c.VeterinarioId)
                .OnDelete(DeleteBehavior.Restrict);

            modelBuilder.Entity<HistoricoClinico>()
                .HasOne(h => h.Pet)
                .WithMany(p => p.Historicos)
                .HasForeignKey(h => h.PetId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Usuario>()
                .HasIndex(u => u.Email)
                .IsUnique()
                .HasDatabaseName("UK_USUARIO_EMAIL");

            modelBuilder.Entity<Usuario>()
                .HasIndex(u => u.ResponsavelId)
                .IsUnique()
                .HasDatabaseName("UK_USUARIO_RESPONSAVEL");

            modelBuilder.Entity<Usuario>()
                .HasIndex(u => u.VeterinarioId)
                .IsUnique()
                .HasDatabaseName("UK_USUARIO_VETERINARIO");

                .WithOne(r => r.Usuario)

    public class ExternalServiceHealthCheck : IHealthCheck
    {
        private readonly HttpClient _httpClient;
        private readonly IConfiguration _configuration;

        public ExternalServiceHealthCheck(HttpClient httpClient, IConfiguration configuration)
        {
            _httpClient = httpClient;
            _configuration = configuration;
        }

        public async Task<HealthCheckResult> CheckHealthAsync(
            HealthCheckContext context,
            CancellationToken cancellationToken = default)
        {
            var url = _configuration["HealthChecks:ExternalServiceUrl"];
            if (string.IsNullOrWhiteSpace(url))
            {
                return HealthCheckResult.Healthy("Nenhum serviÃ§o externo configurado para checagem.");
            }

            var stopwatch = Stopwatch.StartNew();
            try
            {
                using var response = await _httpClient.GetAsync(url, cancellationToken);
                stopwatch.Stop();

                var data = new Dictionary<string, object>
                {
                    ["url"] = url,
                    ["statusCode"] = (int)response.StatusCode,
                    ["responseTimeMs"] = stopwatch.ElapsedMilliseconds
                };

                return response.IsSuccessStatusCode
                    ? HealthCheckResult.Healthy($"ServiÃ§o externo respondeu {(int)response.StatusCode}.", data)
                    : HealthCheckResult.Degraded($"ServiÃ§o externo respondeu {(int)response.StatusCode}.", data: data);
            }
            catch (Exception ex)
            {
                return HealthCheckResult.Unhealthy($"NÃ£o foi possÃ­vel acessar {url}.", ex);
            }
        }
    }

    {
        public static Task WriteResponse(HttpContext context, HealthReport report)
        {
            context.Response.ContentType = "application/json; charset=utf-8";

            var payload = new
            {
                status = report.Status.ToString(),
                totalDurationMs = report.TotalDuration.TotalMilliseconds,
                checks = report.Entries.Select(e => new

                    name = e.Key,
                    status = e.Value.Status.ToString(),
                    description = e.Value.Description,
                    durationMs = e.Value.Duration.TotalMilliseconds,
                    error = e.Value.Exception?.Message,

                })
            };

            var json = JsonSerializer.Serialize(payload, new JsonSerializerOptions
            {
                WriteIndented = true
            });

            return context.Response.WriteAsync(json);
        }

    public class OracleDatabaseHealthCheck : IHealthCheck
    {

        public OracleDatabaseHealthCheck(AppDbContext context)
        {

        }

        public async Task<HealthCheckResult> CheckHealthAsync(
            HealthCheckContext context,
            CancellationToken cancellationToken = default)
        {
            var stopwatch = Stopwatch.StartNew();
            try
            {
                var podeConectar = await _context.Database.CanConnectAsync(cancellationToken);
                if (!podeConectar)
                {
                    return HealthCheckResult.Unhealthy("NÃ£o foi possÃ­vel abrir conexÃ£o com o Oracle.");
                }

                await _context.Database.ExecuteSqlRawAsync("SELECT 1 FROM DUAL", cancellationToken);

                stopwatch.Stop();
                var data = new Dictionary<string, object>
                {
                    ["responseTimeMs"] = stopwatch.ElapsedMilliseconds,
                    ["dataSource"] = _context.Database.GetDbConnection().DataSource
                };

                return stopwatch.ElapsedMilliseconds > 1000
                    ? HealthCheckResult.Degraded("Oracle respondeu, mas acima de 1s.", data: data)
                    : HealthCheckResult.Healthy("ConexÃ£o com o Oracle OK.", data);
            }
            catch (Exception ex)
            {
                return HealthCheckResult.Unhealthy("Falha ao conectar no Oracle.", ex);
            }
        }

namespace YourPetHealth.Middleware
{
    
    public class CorrelationIdMiddleware
    {
        private const string HeaderName = "X-Correlation-Id";
        private readonly RequestDelegate _next;

        public CorrelationIdMiddleware(RequestDelegate next)
        {
            _next = next;
        }

        public async Task InvokeAsync(HttpContext context)
        {
            var correlationId = context.Request.Headers.TryGetValue(HeaderName, out var existing)
                ? existing.ToString()
                : Guid.NewGuid().ToString("N");

            context.Response.Headers[HeaderName] = correlationId;

            using (LogContext.PushProperty("CorrelationId", correlationId))
            {
                await _next(context);
            }
        }
    }

    }

using Microsoft.Extensions.Configuration;
using MongoDB.Driver;

namespace YourPetHealth.Services
{
    public class MongoDbService
    {
        private readonly IMongoDatabase _database;

        public MongoDbService(IConfiguration configuration)
        {
           
            var connectionString = configuration["MongoDbSettings:ConnectionString"] ?? "mongodb://localhost:27017";
            var databaseName = configuration["MongoDbSettings:DatabaseName"] ?? "clyvo_vet_db";

            var client = new MongoClient(connectionString);
            _database = client.GetDatabase(databaseName);
        }

        public IMongoCollection<T> GetCollection<T>(string name)
        {
            return _database.GetCollection<T>(name);
        }
    }
}
